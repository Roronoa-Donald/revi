def gauss(a, b):
    n = len(a)
    for col in range(n):
        pivot = max(range(col, n), key=lambda r: abs(a[r][col]))
        a[col], a[pivot] = a[pivot], a[col]
        b[col], b[pivot] = b[pivot], b[col]
        for r in range(col + 1, n):
            f = a[r][col] / a[col][col]
            for c in range(col, n):
                a[r][c] -= f * a[col][c]
            b[r] -= f * b[col]
    x = [0] * n
    for i in range(n - 1, -1, -1):
        x[i] = (b[i] - sum(a[i][j] * x[j] for j in range(i + 1, n))) / a[i][i]
    return x

def rang(m):
    m = [ligne[:] for ligne in m]
    lignes, cols, r = len(m), len(m[0]), 0
    for c in range(cols):
        piv = next((i for i in range(r, lignes) if m[i][c] != 0), None)

def valeurs_propres(a, b, c, d):
    s, p = a + d, a * d - b * c
    disc = s * s - 4 * p
    if disc < 0:
        return "pas de valeur propre reelle"
    return sorted([(s - disc ** 0.5) / 2, (s + disc ** 0.5) / 2])

Projet intégrateur — « boîte à outils matricielle »

Cahier des charges

Un professeur veut un petit programme Python qui vérifie ses énoncés d’algèbre linéaire. Le programme lit une matrice carrée (première ligne : sa taille n ; puis n lignes de n entiers), puis affiche un rapport complet : trace, déterminant, rang, et — si la matrice est inversible — le fait qu’elle l’est.

Fonctionnalités attendues
  1. Lire la matrice depuis l’entrée standard (une ligne = une ligne de la matrice).
  2. Calculer la trace avec une fonction dédiée (matrices).
  3. Calculer le déterminant par développement récursif ou par échelonnement (déterminants).
  4. Calculer le rang par échelonnement avec pivot (familles et rang).
  5. Conclure : inversible ⇔ rang = n ⇔ det ≠ 0.
  6. Gérer les cas limites : n = 1, matrice nulle, coefficients négatifs.
  7. Bonus : afficher aussi la matrice transposée.

Contraintes : une fonction = une responsabilité, noms explicites, aucun arrondi non demandé.

Grille d’auto-évaluation du projet (sur 100)
Lecture de la matrice et fonctions dédiées 20
Déterminant correct (tests 1×1, 2×2, 3×3) 20
Rang par échelonnement correct 20
Rapport, conclusion, cas limites 20
Lisibilité, noms, découpage 20

Objectif : ≥ 95/100 avant de considérer le projet terminé.

def lire_matrice():
    n = int(input())
    return [[int(x) for x in input().split()] for _ in range(n)]

def trace(m):
    return sum(m[i][i] for i in range(len(m)))

def determinant(m):
    n = len(m)
    if n == 1:
        return m[0][0]
    total = 0
    for j in range(n):
        sous = [ligne[:j] + ligne[j + 1:] for ligne in m[1:]]
        total += ((-1) ** j) * m[0][j] * determinant(sous)
    return total

def rang(m):
    m = [ligne[:] for ligne in m]
    lignes, cols, r = len(m), len(m[0]), 0
    for c in range(cols):
        piv = next((i for i in range(r, lignes) if m[i][c] != 0), None)
        if piv is None:
            continue
        m[r], m[piv] = m[piv], m[r]
        for i in range(r + 1, lignes):
            f = m[i][c] / m[r][c]
            m[i] = [m[i][j] - f * m[r][j] for j in range(cols)]
        r += 1
    return r

matrice = lire_matrice()
n = len(matrice)
print("trace :", trace(matrice))
print("determinant :", determinant(matrice))
print("rang :", rang(matrice))
print("inversible :", "oui" if determinant(matrice) != 0 else "non")
