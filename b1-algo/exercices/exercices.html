def anagrammes(a, b):
    if len(a) != len(b):
        return False
    freq = {}
    for c in a:
        freq[c] = freq.get(c, 0) + 1
    for c in b:
        if c not in freq:
            return False
        freq[c] -= 1
    return all(v == 0 for v in freq.values())

Projet intégrateur — « Machine à notes »

Cahier des charges

Une école veut un petit programme en ligne de commande pour gérer les notes de sa classe. Le programme lit les données depuis l'entrée standard (format nom note1 note2 note3, une ligne par étudiant, précédées d'un nombre d'étudiants), calcule les moyennes, applique un barème d'admission, affiche un rapport trié et sauvegarde le classement dans classement.txt.

Fonctionnalités attendues
  1. Lire les enregistrements et construire une structure de données adaptée (tableaux).
  2. Calculer la moyenne de chaque étudiant avec une fonction dédiée (fonctions).
  3. Trier du meilleur au moins bon par sélection, sans sort() (algorithmes classiques).
  4. Mentionner : « Admis » si moyenne ≥ 10, « Ajourné » sinon (conditions).
  5. Afficher un rapport lisible et écrire classement.txt.
  6. Gérer les cas limites : aucun étudiant, notes à zéro.
  7. Bonus : moyenne de classe, major, taux d'admission.

Contraintes : noms explicites, une fonction = une responsabilité, aucune boucle infinie.

Grille d'auto-évaluation du projet (sur 100)
Lecture des enregistrements et structure des données 20 %
Moyennes via fonctions dédiées 20 %
Tri maison correct (sans sort) 20 %
Rapport, écriture fichier, cas limites 20 %
Lisibilité, noms, découpage 20 %

Objectif : ≥ 95/100 avant de considérer le projet terminé.

def lire_etudiants(n):
    """Retourne une liste [(nom, [notes])] depuis l'entrée standard."""
    etudiants = []
    for i in range(n):
        morceaux = input().split()
        nom = morceaux[0]
        notes = [int(v) for v in morceaux[1:]]
        etudiants.append((nom, notes))
    return etudiants

def moyenne(notes):
    """Moyenne d'une liste de notes (0 si vide)."""
    if not notes:
        return 0.0
    return sum(notes) / len(notes)

def trier_decroissant(etudiants):
    """Tri par sélection sur la moyenne, du meilleur au moins bon."""
    t = list(etudiants)
    for debut in range(len(t) - 1):
        indice_max = debut
        for i in range(debut + 1, len(t)):
            if moyenne(t[i][1]) > moyenne(t[indice_max][1]):
                indice_max = i
        t[debut], t[indice_max] = t[indice_max], t[debut]
    return t

def rapport(etudiants):
    lignes = []
    for nom, notes in etudiants:
        moy = moyenne(notes)
        mention = "Admis" if moy >= 10 else "Ajourne"
        lignes.append(f"{nom} {moy:.2f} {mention}")
    return lignes

n = int(input())
etudiants = lire_etudiants(n)
classes = trier_decroissant(etudiants)

for ligne in rapport(classes):
    print(ligne)

with open("classement.txt", "w", encoding="utf-8") as f:
    for ligne in rapport(classes):
        f.write(ligne + "\n")
