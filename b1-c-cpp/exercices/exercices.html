double moyenne(int t[], int n) {
    int total = 0;
    for (int i = 0; i < n; i++) total += t[i];
    return n > 0 ? (double)total / n : 0.0;
}

int compter_voyelles(const char mot[]) {
    int compte = 0;
    for (int i = 0; mot[i] != '\0'; i++) {
        char c = tolower((unsigned char)mot[i]);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || c == 'y') compte++;
    }
    return compte;
}

void echanger(int *a, int *b) {
    int tmp = *a; *a = *b; *b = tmp;
}

#include <string.h>

int anagrammes(const char *a, const char *b) {
    if (strlen(a) != strlen(b)) return 0;
    int freq[256] = {0};
    for (int i = 0; a[i]; i++) freq[(unsigned char)a[i]]++;
    for (int i = 0; b[i]; i++) freq[(unsigned char)b[i]]--;
    for (int i = 0; i < 256; i++) if (freq[i] != 0) return 0;
    return 1;
}

void inverser(int t[], int n) {
    for (int i = 0, j = n - 1; i < j; i++, j--) {
        int tmp = t[i]; t[i] = t[j]; t[j] = tmp;
    }
}

Projet intégrateur — « Carnet de notes en C »

Cahier des charges

Une école veut un petit programme en ligne de commande pour gérer les notes de sa classe. Le programme lit les données depuis etudiants.csv (format nom,note1,note2,...), calcule les moyennes, applique un barème d'admission, affiche un rapport et sauvegarde les résultats.

Fonctionnalités attendues
  1. Lire etudiants.csv (fichiers) et remplir un tableau de structures (structures).
  2. Calculer la moyenne de chaque étudiant par une fonction dédiée.
  3. Classer du meilleur au moins bon (tri).
  4. Mentionner : « Admis » si moyenne ≥ 10, « Ajourné » sinon (conditions).
  5. Afficher un rapport lisible et écrire resultats.txt.
  6. Gérer l'absence du fichier (retour de fopen à NULL).
  7. Bonus : afficher la moyenne de classe, le major et le taux d'admission.

Contraintes : noms explicites, une fonction = une responsabilité, aucun dépassement de tableau.

Grille d'auto-évaluation du projet (sur 100)
Lecture du fichier robuste 20 %
Structures et calcul des moyennes 20 %
Tri et rapport lisible 20 %
Gestion de l'absence de fichier / cas limites 20 %
Lisibilité, noms, découpage en fonctions 20 %

Objectif : ≥ 95/100 avant de considérer le projet terminé.

Voir une solution de référence (après avoir essayé !) 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_ETUDIANTS 100
#define MAX_NOTES 10

typedef struct {
    char nom[50];
    int notes[MAX_NOTES];
    int nbNotes;
    double moyenne;
} Etudiant;

int lire_etudiants(const char *chemin, Etudiant etudiants[], int max) {
    FILE *f = fopen(chemin, "r");
    if (f == NULL) {
        printf("Fichier %s introuvable.\n", chemin);
        return 0;
    }
    int n = 0;
    char ligne[512];
    while (n < max && fgets(ligne, sizeof(ligne), f)) {
        char *token = strtok(ligne, ",");
        if (token == NULL) continue;
        strncpy(etudiants[n].nom, token, 49);
        etudiants[n].nom[49] = '\0';

        etudiants[n].nbNotes = 0;
        while ((token = strtok(NULL, ",")) != NULL && etudiants[n].nbNotes < MAX_NOTES) {
            etudiants[n].notes[etudiants[n].nbNotes++] = atoi(token);
        }
        n++;
    }
    fclose(f);
    return n;
}

double moyenne_de(const Etudiant *e) {
    if (e->nbNotes == 0) return 0.0;
    int total = 0;
    for (int i = 0; i < e->nbNotes; i++) total += e->notes[i];
    return (double)total / e->nbNotes;
}

void classer(Etudiant etudiants[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = i + 1; j < n; j++) {
            if (etudiants[j].moyenne > etudiants[i].moyenne) {
                Etudiant tmp = etudiants[i];
                etudiants[i] = etudiants[j];
                etudiants[j] = tmp;
            }
        }
    }
}

int main(void) {
    Etudiant etudiants[MAX_ETUDIANTS];
    int n = lire_etudiants("etudiants.csv", etudiants, MAX_ETUDIANTS);
    if (n == 0) return 1;

    for (int i = 0; i < n; i++) etudiants[i].moyenne = moyenne_de(&etudiants[i]);
    classer(etudiants, n);

    FILE *out = fopen("resultats.txt", "w");
    if (out == NULL) return 1;

    int admis = 0;
    for (int i = 0; i < n; i++) {
        const char *mention = etudiants[i].moyenne >= 10 ? "Admis" : "Ajourne";
        if (etudiants[i].moyenne >= 10) admis++;
        printf("%-10s %6.2f  %s\n", etudiants[i].nom, etudiants[i].moyenne, mention);
        fprintf(out, "%s : %.2f - %s\n", etudiants[i].nom, etudiants[i].moyenne, mention);
    }
    printf("Taux d'admission : %.1f %%\n", admis * 100.0 / n);
    fclose(out);
    return 0;
}
