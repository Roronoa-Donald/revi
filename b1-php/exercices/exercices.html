function estPair($n)
{
    return $n % 2 === 0;
}

function moyenne($notes)
{
    if (count($notes) === 0) {
        return 0;
    }
    return array_sum($notes) / count($notes);
}

function anagrammes($a, $b)
{
    if (strlen($a) !== strlen($b)) {
        return false;
    }
    $freq = [];
    foreach (str_split($a) as $c) {
        $freq[$c] = ($freq[$c] ?? 0) + 1;
    }
    foreach (str_split($b) as $c) {
        if (!isset($freq[$c]) || $freq[$c] === 0) {
            return false;
        }
        $freq[$c]--;
    }
    return true;
}

Projet intégrateur — « Mini-boutique en ligne de commande »

Cahier des charges

Une boutique veut un petit script PHP pour préparer ses livraisons. Le programme lit les données depuis l'entrée standard (format article quantite prixBillet — en fait article quantite prix, une ligne par référence, précédées d'un nombre de références), calcule le panier, applique les règles de livraison, affiche un rapport trié et prépare la structure d'un futur INSERT PDO.

Fonctionnalités attendues
  1. Lire les références et construire un tableau associatif article => [quantite, prix].
  2. Calculer le sous-total de chaque référence avec une fonction dédiée (fonctions).
  3. Trier les références par sous-total décroissant, sans sort() ni usort() (tableaux).
  4. Livraison : offerte à partir de 100 EUR, sinon 7 EUR (conditions).
  5. Afficher un rapport lisible et valider les quantités (0 à 1000) avant un futur INSERT.
  6. Gérer les cas limites : aucune référence, quantité nulle.
  7. Bonus : panier moyen, référence la plus chère, taux de TVA.

Contraintes : noms explicites, une fonction = une responsabilité, aucune variable non initialisée.

Grille d'auto-évaluation du projet (sur 100)
Lecture des références et tableau associatif 20 %
Sous-totaux via fonctions dédiées 20 %
Tri maison correct (sans sort/usort) 20 %
Rapport, livraison, validation des données 20 %
Lisibilité, noms, découpage 20 %

Objectif : ≥ 95/100 avant de considérer le projet terminé.

<?php
function lireStock($n)
{
    $stock = [];
    for ($i = 0; $i < $n; $i++) {
        $parts = explode(' ', trim(fgets(STDIN)));
        $stock[$parts[0]] = [(int)$parts[1], (int)$parts[2]];
    }
    return $stock;
}

function sousTotal($ligne)
{
    [$quantite, $prix] = $ligne;
    return $quantite * $prix;
}

function trierParMontant($stock)
{
    $refs = array_keys($stock);
    for ($debut = 0; $debut < count($refs) - 1; $debut++) {
        $indiceMax = $debut;
        for ($i = $debut + 1; $i < count($refs); $i++) {
            if (sousTotal($stock[$refs[$i]]) > sousTotal($stock[$refs[$indiceMax]])) {
                $indiceMax = $i;
            }
        }
        [$refs[$debut], $refs[$indiceMax]] = [$refs[$indiceMax], $refs[$debut]];
    }
    return $refs;
}

function livrable($panier)
{
    return $panier >= 100;
}

$n = (int)trim(fgets(STDIN));
$stock = lireStock($n);

$panier = 0;
foreach ($stock as $ligne) {
    $panier += sousTotal($ligne);
}

$verdict = livrable($panier) ? "Livraison : offerte" : "Livraison : 7 EUR";
$total = $panier + (livrable($panier) ? 0 : 7);
echo $verdict . "\n";
echo "Total : $total EUR\n";

foreach (trierParMontant($stock) as $ref) {
    [$quantite, $prix] = $stock[$ref];
    $valide = ($quantite >= 0 && $quantite <= 1000) ? "ok" : "HORS BORNES";
    echo "$ref : $quantite x $prix = " . sousTotal($stock[$ref]) . " EUR [$valide]\n";
}
