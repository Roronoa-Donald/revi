def mediane(valeurs):
    v = sorted(valeurs)
    n = len(v)
    return v[n // 2] if n % 2 else (v[n // 2 - 1] + v[n // 2]) / 2

def quantile(valeurs, p):
    v = sorted(valeurs)
    r = (len(v) + 1) * p
    bas = int(r)
    if bas == r:
        return v[bas - 1]
    return v[bas - 1] + (r - bas) * (v[bas] - v[bas - 1])

def corrélation(xs, ys):
    n = len(xs)
    mx, my = sum(xs) / n, sum(ys) / n
    cov = sum((xs[i] - mx) * (ys[i] - my) for i in range(n)) / n
    sx = (sum((x - mx) ** 2 for x in xs) / n) ** 0.5
    sy = (sum((y - my) ** 2 for y in ys) / n) ** 0.5
    return cov / (sx * sy)

def regression(xs, ys):
    n = len(xs)
    mx, my = sum(xs) / n, sum(ys) / n
    cov = sum((xs[i] - mx) * (ys[i] - my) for i in range(n)) / n
    vx = sum((x - mx) ** 2 for x in xs) / n
    a = cov / vx
    return a, my - a * mx      # y = a·x + b

def classes(mini, maxi, k):
    pas = (maxi - mini) / k
    return [(mini + i * pas, mini + (i + 1) * pas) for i in range(k)]

Projet intégrateur — « rapport d’analyse d’une série »

Cahier des charges

Un service RH te confie les notes obtenues par une promotion à un test de recrutement. Le programme lit n, puis les n notes (une par ligne), et affiche un rapport complet sur la distribution : effectif, étendue, moyenne, médiane, quartiles, variance, écart-type, coefficient de variation et une appréciation de la dispersion.

Fonctionnalités attendues
  1. Lire la série et la trier une seule fois.
  2. Calculer les indicateurs avec des fonctions dédiées (tendance centrale, dispersion).
  3. Interpréter : afficher « dispersion forte » si le coefficient de variation dépasse 25 %, « dispersion modérée » sinon.
  4. Gérer les cas limites : n = 1, série constante (écart-type nul), valeurs négatives.
  5. Afficher deux décimales partout, sauf le CV exprimé en pourcentage.
  6. Bonus : détecter les valeurs aberrantes hors de l’intervalle [Q1 − 1,5·EI ; Q3 + 1,5·EI].

Contraintes : une fonction = une responsabilité, noms explicites, aucune division par zéro.

import math

def moyenne(valeurs):
    return sum(valeurs) / len(valeurs)

def quantile(valeurs, p):
    r = (len(valeurs) + 1) * p
    bas = int(r)
    if bas == r:
        return valeurs[bas - 1]
    return valeurs[bas - 1] + (r - bas) * (valeurs[bas] - valeurs[bas - 1])

def ecart_type(valeurs):
    m = moyenne(valeurs)
    return math.sqrt(sum((x - m) ** 2 for x in valeurs) / len(valeurs))

def coefficient_variation(valeurs):
    m = moyenne(valeurs)
    return ecart_type(valeurs) / m if m != 0 else 0

n = int(input())
serie = sorted(float(input()) for _ in range(n))

m = moyenne(serie)
med = quantile(serie, 0.5)
q1 = quantile(serie, 0.25)
q3 = quantile(serie, 0.75)
s = ecart_type(serie)
cv = coefficient_variation(serie)

print(f"effectif : {n}")
print(f"etendue : {serie[-1] - serie[0]:.2f}")
print(f"moyenne : {m:.2f}")
print(f"mediane : {med:.2f}")
print(f"Q1 : {q1:.2f}   Q3 : {q3:.2f}")
print(f"variance : {s * s:.2f}")
print(f"ecart-type : {s:.2f}")
print(f"coefficient de variation : {cv * 100:.2f} %")
print("dispersion forte" if cv > 0.25 else "dispersion moderee")
