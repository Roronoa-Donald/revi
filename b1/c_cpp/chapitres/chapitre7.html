Chapitre 7

⚡ Gestion des Fichiers

Lire et écrire des fichiers texte et binaire : fopen, fclose, fprintf, fscanf, fgets — la persistance des données en C.

Objectifs du chapitre

  • Créer des structures (struct) et des typedef
  • Lire et écrire dans des fichiers avec FILE*
  • Maîtriser fopen, fclose, fprintf, fscanf

7.1 — Pourquoi les fichiers ?

Analogie — Le cahier de notes

Les variables sont comme un tableau blanc : dès qu'on éteint (ferme le programme), tout s'efface. Un fichier est comme un cahier : ce que vous écrivez reste même après avoir fermé le cahier. C'est la persistance.

📁 Types de fichiers en C

  • Fichier texte — lisible par un humain (.txt, .csv). Contient des caractères ASCII.
  • Fichier binaire — lisible par la machine. Plus compact, plus rapide, mais illisible.

7.2 — Ouvrir et fermer un fichier

fopen et fclose
FILE *f;  // Pointeur vers le fichier

// Ouvrir un fichier
f = fopen("donnees.txt", "r");  // "r" = lecture

// TOUJOURS vérifier si l'ouverture a réussi !
if (f == NULL) {
    printf("Erreur : fichier introuvable !\n");
    return 1;
}

// ... utiliser le fichier ...

// Fermer le fichier (OBLIGATOIRE)
fclose(f);
ModeSignificationSi fichier n'existe pas

"w"Write — écriture (écrase !)Crée le fichier
"a"Append — ajout à la finCrée le fichier

Mnémonique — R.W.A. (Read, Write, Append)
  • r = Read (lire)
  • w = Write (écrire — ⚠️ EFFACE le contenu existant !)
  • a = Append (ajouter à la fin — contenu préservé)
⚠️ Le piège de "w" : Ouvrir un fichier en mode "w" efface immédiatement tout son contenu ! Si vous voulez ajouter sans effacer, utilisez "a".

7.3 — Écrire dans un fichier

fprintf, fputs, fputc
FILE *f = fopen("resultat.txt", "w");
if (f == NULL) return 1;

// fprintf : comme printf mais dans un fichier
fprintf(f, "Nom : %s\n", "Alice");
fprintf(f, "Note : %.2f\n", 15.5);
fprintf(f, "Date : %02d/%02d/%d\n", 15, 3, 2024);

// fputs : écrire une chaîne (sans formatage)
fputs("Fin du rapport\n", f);

// fputc : écrire un seul caractère
fputc('X', f);

fclose(f);
printf("Fichier écrit avec succès !\n");
Astuce : fprintf(f, ...) a exactement la même syntaxe que printf(...), sauf qu'on ajoute f en premier paramètre. Tout ce que vous savez faire avec printf, vous pouvez le faire avec fprintf !

7.4 — Lire un fichier

fscanf, fgets, fgetc
FILE *f = fopen("donnees.txt", "r");
if (f == NULL) return 1;

// Méthode 1 : fgets — lire ligne par ligne (RECOMMANDÉ)
char ligne[256];
while (fgets(ligne, 256, f) != NULL) {
    printf("%s", ligne);  // Affiche chaque ligne
}

// Rembobiner pour relire
rewind(f);

// Méthode 2 : fscanf — lire des données formatées
char nom[50];
int age;
while (fscanf(f, "%s %d", nom, &age) == 2) {
    printf("%s a %d ans\n", nom, age);
}

// Méthode 3 : fgetc — lire caractère par caractère
rewind(f);
int c;
while ((c = fgetc(f)) != EOF) {
    putchar(c);  // Affiche chaque caractère
}

fclose(f);
Mnémonique — Quand utiliser quoi ?
  • fgets → Lire ligne par ligne (le plus sûr et courant)
  • fscanf → Lire des données structurées (nom age note…)
  • fgetc → Lire caractère par caractère (rare)
EOF = End Of File : La constante EOF (définie dans stdio.h) signale qu'on a atteint la fin du fichier. C'est le signal d'arrêt de lecture.

7.5 — Exemple complet : fichier CSV

Écrire et lire un fichier CSV
typedef struct {
    char nom[30];
    float note;
} Etudiant;

// ÉCRITURE
void sauvegarder(Etudiant promo[], int n) {
    FILE *f = fopen("notes.csv", "w");
    if (!f) return;
    fprintf(f, "Nom,Note\n");  // En-tête CSV
    for (int i = 0; i < n; i++)
        fprintf(f, "%s,%.2f\n", promo[i].nom, promo[i].note);
    fclose(f);
}

// LECTURE
int charger(Etudiant promo[], int maxN) {
    FILE *f = fopen("notes.csv", "r");
    if (!f) return 0;
    char ligne[100];
    fgets(ligne, 100, f);  // Sauter l'en-tête
    int n = 0;
    while (n < maxN && fscanf(f, "%[^,],%f\n", 
            promo[n].nom, &promo[n].note) == 2) {
        n++;
    }
    fclose(f);
    return n;
}

int main() {
    Etudiant p[] = {{"Alice",16},{"Bob",12},{"Eve",15}};
    sauvegarder(p, 3);
    
    Etudiant loaded[10];
    int n = charger(loaded, 10);
    for (int i = 0; i < n; i++)
        printf("%s : %.2f\n", loaded[i].nom, loaded[i].note);
    return 0;
}

7.6 — Fichiers binaires (aperçu)

fwrite et fread
Etudiant promo[3] = {{"Alice",16},{"Bob",12},{"Eve",15}};

// Écriture binaire
FILE *f = fopen("promo.dat", "wb");
fwrite(promo, sizeof(Etudiant), 3, f);
fclose(f);

// Lecture binaire
Etudiant loaded[3];
f = fopen("promo.dat", "rb");
fread(loaded, sizeof(Etudiant), 3, f);
fclose(f);

// Avantage : un seul appel pour tout écrire/lire
// Inconvénient : fichier non lisible par un humain

Exercices guidés pas à pas

✏️ Exercice guidé — Sauvegarder des notes dans un fichier

Objectif : Écrire et relire des données dans un fichier texte.

Étape 1 — Écrire dans le fichier :

FILE *f = fopen("notes.txt", "w");
if (f == NULL) {
    printf("Erreur d'ouverture !\n");
    return 1;
}

fprintf(f, "Alice,15\n");
fprintf(f, "Bob,18\n");
fprintf(f, "Clara,12\n");
fclose(f);  // Toujours fermer !

Étape 2 — Relire le fichier :

FILE *f = fopen("notes.txt", "r");
char nom[50];
int note;

while (fscanf(f, "%[^,],%d\n", nom, &note) == 2) {
    printf("%-10s : %d/20\n", nom, note);
}
fclose(f);
💡 %[^,] lit tous les caractères sauf la virgule — très utile pour parser du CSV !

✏️ Exercice guidé — Copier un fichier

Objectif : Copier le contenu d'un fichier dans un autre, caractère par caractère.

FILE *src = fopen("source.txt", "r");
FILE *dst = fopen("copie.txt", "w");

if (src == NULL || dst == NULL) {
    printf("Erreur d'ouverture !\n");
    return 1;
}

int c;
while ((c = fgetc(src)) != EOF) {
    fputc(c, dst);
}

fclose(src);
fclose(dst);
printf("Copie terminée !\n");
⚠️ Le type de c doit être int (pas char) car EOF est un entier (-1) !
