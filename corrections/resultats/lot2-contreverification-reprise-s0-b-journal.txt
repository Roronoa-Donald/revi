import json
from decimal import Decimal
from pathlib import Path


exercise = json.loads(Path('/travail/f221-exercise.json').read_text())
assert exercise['id'] == 'vrt3-c5-e1' and exercise['language'] == 'analyse'


def compute(line):
    values = dict(part.split('=', 1) for part in line.split())
    network = Decimal(values['NET_GBPS']) * Decimal(1000) * Decimal('0.8') / Decimal(8)
    dirty = Decimal(values['DIRTY_RATE_MBPS'])
    memory = Decimal(values['RAM_GB']) * Decimal(1000)
    if dirty >= network:
        transferred, duration, status = Decimal(0), Decimal(0), 'CONVERGENCE_FAILURE'
    else:
        first_duration = memory / network
        second_memory = dirty * first_duration
        second_duration = second_memory / network
        transferred = memory + second_memory
        duration = first_duration + second_duration
        status = 'RAPIDE' if duration <= 30 else 'ACCEPTABLE' if duration <= 90 else 'LENT'
    answer = f"VM:{values['VM']} | NET:{values['NET_GBPS']}Gbps | TRANSFERRED:{transferred:.1f}MB | DURATION:{duration:.1f}s | STATUT:{status}"
    return {'input': line, 'network_MB_s': str(network), 'duration_raw_s': str(duration), 'answer': answer}


starter = exercise['starter'].splitlines()
cases = []
for line in starter:
    if line.startswith('[Cas '):
