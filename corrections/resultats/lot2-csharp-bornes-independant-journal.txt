Le banc calcule les attendus indépendants en entiers arbitraires, contrôle les
deux nouveaux cas dans la tranche effective MAX_TESTS et vérifie la route
suspendue. Pour les demi-centimes, il compare séparément les arrondis au pair
et à l'écart de zéro, car l'énoncé ne précise pas le mode d'arrondi. La notation exponentielle est explicitement
hors domaine dans la dernière version ; elle est explorée séparément et n'est
pas comptée comme défaut. Le cumul est borné par l'environnement d'exécution.
Les pages et le catalogue ont changé pendant le démarrage en parallèle ; les
empreintes initiales du banc ci-dessous identifient la version effectivement
testée après la précision de l'utilisateur, puis sont comparées en fin de banc.
Le premier essai n'a pas pu être enregistré à cause de la variable spéciale
en lecture seule « status » du shell zsh. Le banc enregistré ensuite était
incomplet : il supposait à tort 7 tests pour les deux exercices, alors que le
premier en a 6. Son oracle supposait aussi l'arrondi au pair. Ces deux erreurs
de harnais sont conservées dans sa sortie brute ; le banc suivant les corrige.

function normalized(value) { return value.replace(/\r\n?/g, '\n').replace(/\n+$/, ''); }
function priceFixed(raw, rounding = 'even') {
  const negative = raw.startsWith('-');
  const unsigned = /^[+-]/.test(raw) ? raw.slice(1) : raw;
  const [integer = '', fraction = ''] = unsigned.split('.');
  let cents = BigInt(integer || '0') * 100n + BigInt((fraction + '00').slice(0, 2));
  const more = fraction.slice(2);
  if (more && (more[0] > '5' || (more[0] === '5' && (/[1-9]/.test(more.slice(1)) || rounding === 'away' || cents % 2n === 1n)))) cents += 1n;
  return (negative ? '-' : '') + String(cents / 100n) + '.' + String(cents % 100n).padStart(2, '0');
}
function articleLine(item, rounding = 'even') {
  const call = item.raw !== undefined ? `Article("${item.ref}", "${item.name}", ${item.raw})`
    : item.name !== undefined ? `Article("${item.ref}", "${item.name}")` : `Article("${item.ref}")`;
  const chain = item.raw !== undefined ? 'ctor(r, nom, prix)' : item.name !== undefined
    ? 'ctor(r, nom) -> ctor(r, nom, prix)' : 'ctor(r) -> ctor(r, nom) -> ctor(r, nom, prix)';
  const name = item.name === undefined ? 'Sans nom' : item.name;
  const amount = item.raw === undefined ? '0.00' : priceFixed(item.raw, rounding);
  return `${call} => Chaine: ${chain} | Objet: [Ref=${item.ref}, Nom=${name}, Prix=${amount}]`;
}
function timeLines(tasks) {
  let time = 0n;
  const lines = [];
  tasks.forEach((task, index) => {
    lines.push(index === 0 ? `[T=${time}ms] Demarrage ${task.name} -> Await (trace simulee)`
      : `[T=${time}ms] Reprise contexte -> Demarrage ${task.name} -> Await`);
    time += BigInt(task.duration);
  });
  lines.push(`[T=${time}ms] Tache completee avec succes. Duree totale=${time}ms`);
  return lines.join('\n');
}
