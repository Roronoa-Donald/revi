'use strict';

const assert = require('node:assert/strict');
const crypto = require('node:crypto');

const { runAnalyse } = require('/repo/server/services/execution/local-runner');
const { isGradingSuspended } = require('/repo/server/exercises/grading-policy');

const sha = value => crypto.createHash('sha256').update(value, 'utf8').digest('hex');
const trim = value => String(value).trim();
const compact = value => String(value).replace(/\s+/g, ' ').trim();
const fields = line => Object.fromEntries(line.split(/\s+/).slice(1).filter(token => token.includes('=')).map(token => {
  const pos = token.indexOf('=');
  assert.ok(pos > 0, `champ illisible : ${token}`);
  return [token.slice(0, pos), token.slice(pos + 1)];
}));

function dataCases(exercise) {
  const result = [];
  let current = null;
  for (const line of exercise.starter.split('\n')) {
    const header = /^\[Cas (\d+)(?:\s|\])/.exec(line);
    if (header) {
      current = { number: Number(header[1]), lines: [] };
      result.push(current);
    } else if (current && line.startsWith('# Données : ')) {
      current.lines.push(line.slice('# Données : '.length));
    }
  }
  assert.equal(result.length, exercise.tests.length, `${exercise.id} : nombre de cas`);
  return result;
}

function allNodes(node) {
  return [node, ...(node.childNodes || []).flatMap(allNodes)];
}
const nodeText = node => allNodes(node).filter(child => child.nodeName === '#text').map(child => child.value).join(' ');
const attribute = (node, name) => (node.attrs || []).find(item => item.name === name)?.value;

function pageCheck(exercise) {
  const filename = path.join('/repo', exercise.course, 'chapitres', `chapitre${exercise.chapter}.html`);
  const document = parse5.parse(fs.readFileSync(filename, 'utf8'));
  const labs = allNodes(document).filter(node => attribute(node, 'data-exercise') === exercise.id);
  assert.equal(labs.length, 1, `${exercise.id} : atelier public unique`);
  const lab = labs[0];
  assert.equal(attribute(lab, 'data-language'), exercise.language);
  assert.equal(attribute(lab, 'data-stdin'), exercise.examples[0].stdin);
  const starter = allNodes(lab).find(node => attribute(node, 'data-lab-starter') !== undefined);
  assert.ok(starter, `${exercise.id} : éditeur public`);
  assert.equal((starter.childNodes || []).map(node => node.value || '').join(''), exercise.starter);
  const brief = allNodes(lab).find(node => (attribute(node, 'class') || '').split(' ').includes('lab__brief'));
  assert.ok(brief, `${exercise.id} : énoncé public`);
  assert.equal(compact(nodeText(brief)), compact(nodeText(parse5.parseFragment(exercise.brief))));
  const publicView = toClient(exercise);
  assert.ok(!Object.hasOwn(publicView, 'tests') && !Object.hasOwn(publicView, 'solution'));
  return filename.slice('/repo/'.length);
}

async function grade(exercise, replies) {
  const answer = replies.map((reply, index) => `[Cas ${index + 1}]\n${reply}`).join('\n\n');
  return runTests(exercise, answer, {
    maxTests: MAX_TESTS,
    execute: async ({ code, stdin }) => {
      const output = await runAnalyse(code, stdin);
      return { ok: output.ok, status: output.ok ? 'success' : 'runtime_error', stdout: output.data.stdout,
        stderr: output.data.stderr, exitCode: output.data.exit_code };
    }
  });
}

const oracles = {};

oracles['st-c2-e2'] = lines => {
  const values = lines.map(Number);
  const count = values.shift();
  assert.equal(values.length, count);
  const frequencies = new Map();
  for (const value of values) frequencies.set(value, (frequencies.get(value) || 0) + 1);
  return [...frequencies].sort(([left], [right]) => left - right)
    .map(([value, amount]) => `${value} ${(100 * amount / count).toFixed(2)}`).join('\n');
};

oracles['eq3-c4-e3'] = lines => {
  const data = fields(lines[0]);
  const [oo, on, no, nn] = ['OO', 'ON', 'NO', 'NN'].map(key => Number(data[key]));
  const total = oo + on + no + nn;
  const agreement = (oo + nn) / total;
  const expected = ((oo + on) * (oo + no) + (no + nn) * (on + nn)) / total ** 2;
  const kappa = expected === 1 ? 'INDEFINI' : ((agreement - expected) / (1 - expected)).toFixed(3);
  return `ACCORD_OBSERVATEURS: ${data.session} -> ACCORDS=${oo + nn}/${total} (${(100 * agreement).toFixed(1)}%) | PE=${(100 * expected).toFixed(1)}% | KAPPA=${kappa}`;
};

oracles['eq3-c2-e3'] = lines => {
  const data = fields(lines[0]);
  const split = value => Object.fromEntries(value.split(',').map(item => item.split(':')).map(([key, count]) => [key, Number(count)]));
  const targets = split(data.cibles);
  const observed = split(data.obs);
  const total = Object.values(observed).reduce((sum, value) => sum + value, 0);
  const delta = Math.max(...Object.keys(targets).map(key => Math.abs(100 * observed[key] / total - targets[key])));
  return `AUDIT_QUOTAS: ${data.etude} -> TOTAL_OBS=${total} | ECART_MAX=${delta.toFixed(1)} points | CONFORMITE_QUOTAS=${delta <= 3 ? 'OUI' : 'NON'}`;
};

function modPow(value, exponent, modulo) {
  let result = 1n;
  while (exponent > 0n) {
    if (exponent % 2n) result = (result * value) % modulo;
    value = (value * value) % modulo;
    exponent /= 2n;
  }
  return result;
}

function executeInIsolation(language, code, input) {

  ];

    ? ['/usr/bin/python3', '-I', '-c', code]

    for (const statement of ast.body) {

import ctypes
import os
import socket
import struct


class Filter(ctypes.Structure):
    _fields_ = [('code', ctypes.c_ushort), ('jt', ctypes.c_ubyte), ('jf', ctypes.c_ubyte), ('k', ctypes.c_uint32)]


class Program(ctypes.Structure):
    _fields_ = [('length', ctypes.c_ushort), ('instructions', ctypes.POINTER(Filter))]


def udp():
    endpoint = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    endpoint.bind(('127.0.0.1', 0))
    endpoint.settimeout(0.5)
    return endpoint


def request():
    return struct.pack('!HHI12s', 1, 0, 0x2112A442, os.urandom(12))


def response(message, address):
    encoded_port = address[1] ^ 0x2112
    encoded_ip = int.from_bytes(socket.inet_aton(address[0]), 'big') ^ 0x2112A442
    attribute = struct.pack('!HHBBHI', 0x0020, 8, 0, 1, encoded_port, encoded_ip)
    return struct.pack('!HHI12s', 0x0101, len(attribute), 0x2112A442, message[8:20]) + attribute


def decode(message):
    kind, size, reserved, family, port, address = struct.unpack('!HHBBHI', message[20:32])
    assert (kind, size, reserved, family) == (0x0020, 8, 0, 1)
    return socket.inet_ntoa((address ^ 0x2112A442).to_bytes(4, 'big')), port ^ 0x2112


client = udp()
server_one = udp()
server_two = udp()
peer = udp()
port_one = server_one.getsockname()[1]
port_two = server_two.getsockname()[1]
instructions = (Filter * 5)(Filter(0x28, 0, 0, 0), Filter(0x15, 2, 0, port_one),
                            Filter(0x15, 1, 0, port_two), Filter(0x06, 0, 0, 0),
                            Filter(0x06, 0, 0, 65535))
program = Program(len(instructions), instructions)
libc = ctypes.CDLL(None, use_errno=True)
status = libc.setsockopt(client.fileno(), socket.SOL_SOCKET, 26, ctypes.byref(program), ctypes.sizeof(program))
if status != 0:
    raise OSError(ctypes.get_errno(), 'SO_ATTACH_FILTER')

mappings = []
for server in (server_one, server_two):
    client.sendto(request(), server.getsockname())
    message, source = server.recvfrom(1024)
    server.sendto(response(message, source), source)
    reply, _source = client.recvfrom(1024)
    mappings.append(decode(reply))
assert mappings[0] == mappings[1] == client.getsockname()
peer.sendto(b'paquet-pair', client.getsockname())
try:
    client.recvfrom(1024)
    raise AssertionError('le filtrage du pair a échoué')
except TimeoutError:
    pass
print(f'F180: deux réponses STUN Binding minimales XOR-MAPPED-ADDRESS={mappings[0]} / {mappings[1]}')
print(f'F180: paquet UDP du pair {peer.getsockname()} émis, non reçu dans le délai (filtre noyau cBPF); les réponses des deux serveurs sont reçues')
print('Limite: réseau loopback et filtrage réels mais aucun NAT/TURN ni Internet; serveurs STUN minimaux locaux.')

=== DEBUT SCRIPT tls-wire.py ===
import re
import struct
import sys


def server_hellos(path):
    lines = open(path, encoding='utf8').read().splitlines()
    results = []
    for index, line in enumerate(lines):
        if not line.startswith('<<<') or 'Handshake' not in line or 'ServerHello' not in line:
            continue
        encoded = []
        for following in lines[index + 1:]:
            pieces = following.strip().split()
            if not pieces or any(not re.fullmatch('[0-9a-fA-F]{2}', value) for value in pieces):
                break
            encoded.extend(pieces)
        data = bytes.fromhex(''.join(encoded))
        assert data[0] == 2 and int.from_bytes(data[1:4], 'big') == len(data) - 4
        random = data[6:38]
        offset = 38
        session_length = data[offset]
        offset += 1 + session_length
        cipher = int.from_bytes(data[offset:offset + 2], 'big')
        offset += 3
        length = int.from_bytes(data[offset:offset + 2], 'big')
        offset += 2
        assert offset + length == len(data)
        extensions = []
        while offset < len(data):
            kind, size = struct.unpack('!HH', data[offset:offset + 4])
            offset += 4 + size
            extensions.append(kind)
        results.append((random.hex(), cipher, extensions))
    return results


hrr = server_hellos(sys.argv[1])
psk = server_hellos(sys.argv[2])
assert len(hrr) == 2 and len(psk) == 1
assert hrr[0][0] == 'cf21ad74e59a6111be1d8c021e65b891c2a211167abb8c5e079e09e2c8a8339c'
assert hrr[1][1] == psk[0][1] == 0x1301
assert 51 in hrr[1][2] and 41 not in hrr[1][2]
assert 41 in psk[0][2] and 51 not in psk[0][2]
print('F188: ServerHello TLS_AES_128_GCM_SHA256 avec certificat: extensions=', hrr[1][2], 'key_share présent')
print('F188: même suite avec PSK: extensions=', psk[0][2], 'pre_shared_key présent, key_share absent (psk_ke)')
