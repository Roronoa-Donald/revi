Mémo syntaxe React

Toutes les structures et syntaxes essentielles de React réunies sur une seule page. À garder ouvert pendant vos ateliers pratiques.

1. Fondations JSX & Composants

// Règle 1 : Racine unique ou Fragment
return (
  <>
    <h1>Titre</h1>
    <p>Contenu</p>
  </>
);

// Règle 2 : Attributs en camelCase et balises auto-fermantes
<input className="form-control" type="text" tabIndex={1} />

// Règle 3 : Évaluation d’expressions JavaScript entre accolades
<span>Total : {prix * quantite} €</span>

2. Props & Composition

// Composant avec déstructuration et valeur de repli
function UserBadge({ nom, role = 'visiteur', children }) {
  return (
    <div className="badge">
      <strong>{nom}</strong> ({role})
      <div>{children}</div>
    </div>
  );
}

// Utilisation
<UserBadge nom="Alice" role="admin">
  <p>Droits d’accès complets</p>
</UserBadge>

3. État local (useState) & Formulaires

// Déclaration d’état et setter
const [compteur, setCompteur] = useState(0);
const [texte, setTexte] = useState('');

// Mise à jour basée sur l’état précédent
setCompteur(prev => prev + 1);

// Input contrôlé et soumission sans rechargement
<form onSubmit={(e) => { e.preventDefault(); envoyer(); }}>
  <input value={texte} onChange={(e) => setTexte(e.target.value)} />
</form>

4. Listes (map) & Rendu conditionnel

// Rendu de liste avec prop key unique
<ul>
  {articles.map(article => (
    <li key={article.id}>{article.titre}</li>
  ))}
</ul>

// Rendu conditionnel ternaire et court-circuit
{estConnecte ? <Dashboard /> : <Login />}
{nbAlertes > 0 && <span className="pastille">{nbAlertes}</span>}

5. Effets de bord (useEffect)

// 1. Montage initial uniquement
useEffect(() => {
  console.log('Composant monté');
}, []);

// 2. Synchronisation sur dépendance avec nettoyage
useEffect(() => {
  const timer = setInterval(() => tick(), 1000);
  return () => clearInterval(timer); // Cleanup au démontage
}, [intervalle]);

// 3. Appel API asynchrone sécurisé
useEffect(() => {
  let actif = true;
  async function load() {
    const data = await fetchUser(id);
    if (actif) setUser(data);
  }
  load();
  return () => { actif = false; };
}, [id]);

6. Routage (React Router v6)

import { BrowserRouter, Routes, Route, Link, useParams } from 'react-router-dom';

// Arborescence des routes
<BrowserRouter>
  <nav><Link to="/accueil">Accueil</Link></nav>
  <Routes>
    <Route path="/accueil" element={<Accueil />} />
    <Route path="/article/:id" element={<DetailArticle />} />
  </Routes>
</BrowserRouter>

// Extraction d’un paramètre dynamique dans le composant
function DetailArticle() {
  const { id } = useParams();
  return <h2>Article #{id}</h2>;
}
